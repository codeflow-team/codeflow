  return paths.slice(0, Math.max(0, max));

function countLines(content: string): number {

function extractExportedNames(source: string): string[] {
  const names = new Set<string>();
  const declarations =
    /\bexport\s+(?:declare\s+)?(?:default\s+)?(?:async\s+)?(?:function\s*\*?|class|const|let|var|interface|type|enum)\s+([A-Za-z0-9_$]+)/g;
  const braceLists = /\bexport\s*\{([^}]*)\}/g;
  const starReexports =
    /\bexport\s*\*\s*(?:as\s+[A-Za-z0-9_$]+\s*)?from\s*["']([^"']+)["']/g;

  for (const match of source.matchAll(declarations)) {
    names.add(match[1]);
  }
  for (const match of source.matchAll(braceLists)) {
    for (const part of match[1].split(",")) {
      const name = part.trim().split(/\s+as\s+/)[0].trim();
      if (name.length > 0) {
        names.add(name);
      }
    }
  }
  for (const match of source.matchAll(starReexports)) {
    names.add(`* from "${match[1]}"`);
  }
  return Array.from(names);
}

function extractSizeBytes(fileInfoText: string): number {
  const jsonMatch = fileInfoText.match(/"size"\s*:\s*(\d+)/);
  if (jsonMatch !== null) {
    return Number(jsonMatch[1]);
  }
  const looseMatch = fileInfoText.match(/[Ss]ize\s*[:=]\s*(\d+)/);
  if (looseMatch !== null) {
    return Number(looseMatch[1]);
  }
  return fileInfoText.length;
}

function gradeModule(sizeBytes: number, sourceText: string): string {
  if (sizeBytes > 200) {
    return "large";
  }
  if (sourceText.includes("/**")) {
    return "documented";
  }
  return "plain";
}

function countByGrade(grades: ModuleGrade[]): Record<string, number> {

import type { Tools } from "../generated/tools";

interface BatchNote {
  path: string;
  text: string;
}

interface NoteFailure {
  path: string;
  reason: string;
}

/** Parses the newline-separated path list returned by fs.searchFiles. */
function parseFoundPaths(content: string): string[] {
  return content
    .split("\n")
    .map((line) => line.trim())
    .filter((line) => line !== "");
}

/** Splits the found notes into batches of at most batchSize paths. */
function chunkIntoBatches(paths: string[], batchSize: number): string[][] {
  const size = Math.max(1, Math.floor(batchSize));
  const batches: string[][] = [];
  for (let i = 0; i < paths.length; i += size) {
    batches.push(paths.slice(i, i + size));
  }
  return batches;
}

/**
 * Splits the combined output of one fs.readMultipleFiles call back into
 * per-note texts. The server formats each file as `<path>:\n<text>`,
 * separated by blank lines, so the known batch paths are used as markers.
 */
function splitBatchContent(content: string, paths: string[]): BatchNote[] {
  const notes: BatchNote[] = [];
  for (let i = 0; i < paths.length; i += 1) {
    const marker = `${paths[i]}:\n`;
    const start = content.indexOf(marker);
    if (start === -1) {
      continue;
    }
    const textStart = start + marker.length;
    let textEnd = content.length;
    if (i + 1 < paths.length) {
      const nextStart = content.indexOf(`${paths[i + 1]}:\n`, textStart);
      if (nextStart !== -1) {
        textEnd = nextStart;
      }
    }
    notes.push({
      path: paths[i],
      text: content.slice(textStart, textEnd).replace(/\s+$/, ""),
    });
  }
  return notes;
}

/** Prefixes a note with the migration marker; an empty note stays empty. */
function withMigrationMarker(text: string): string {
  if (text.trim() === "") {
    return "";
  }
  return `<!-- migrated -->\n${text}`;
}

/** Last segment of a path, used as the archived file name. */
function baseName(path: string): string {
  const parts = path.split("/");
  return parts[parts.length - 1];
}

/** Destination path for a note inside <root>/archive. */
function archivePathFor(root: string, originalPath: string): string {
  return `${root}/archive/${baseName(originalPath)}`;
}

/** Records a failed note so the run can carry on with the next one. */
function recordFailure(failures: NoteFailure[], path: string, error: unknown): void {
  failures.push({ path, reason: String(error) });
}

/** Renders the final migration-report.md contents. */
function buildReport(
  migrated: number,
  skipped: number,
  failures: NoteFailure[],
  abandoned: boolean
): string {
  const lines: string[] = [
    "# Migration report",
    "",
    `Migrated: ${migrated}`,
    `Skipped: ${skipped}`,
    `Failed: ${failures.length}`,
    `Abandoned: ${abandoned ? "yes" : "no"}`,
  ];
  if (failures.length > 0) {
    lines.push("", "## Failures");
    for (const failure of failures) {
      lines.push(`- ${failure.path}: ${failure.reason}`);
    }
  }
  return lines.join("\n");
}

export default async function flow(
  input: { root: string; batchSize: number; maxFailures: number },
  tools: Tools
) {
  // 1. Make sure the destination exists before anything is written.
  await tools.fs.createDirectory({ path: `${input.root}/archive` });

  // 2. Find every markdown note under the root.
  const found = await tools.fs.searchFiles({ path: input.root, pattern: "**/*.md" });
  const notePaths = parseFoundPaths(found.content);

  if (notePaths.length === 0) {
    await tools.fs.writeFile({ path: `${input.root}/migration-report.md`, content: "" });
    return { migrated: 0, skipped: 0, failed: 0, abandoned: false };
  }

  const batches = chunkIntoBatches(notePaths, input.batchSize);

  let migrated = 0;
  let skipped = 0;
  let abandoned = false;
  const failures: NoteFailure[] = [];

  // 3. Loop over batches; each batch is read with a single call.
  for (const batch of batches) {
    if (abandoned) {
      break;
    }

    let raw: string | null = null;
    try {
      const batchRead = await tools.fs.readMultipleFiles({ paths: batch });
      raw = batchRead.content;
    } catch (error) {
      for (const path of batch) {
        recordFailure(failures, path, error);
      }
    }

    if (raw === null) {
      if (failures.length > input.maxFailures) {
        abandoned = true;
        break;
      }
      continue;
    }

    const notes = splitBatchContent(raw, batch);

    // Handle each note in the batch one at a time.
    for (const note of notes) {
      const transformed = withMigrationMarker(note.text);

      // 4. An empty note counts as neither a success nor a failure.
      if (transformed === "") {
        skipped += 1;
        continue;
      }

      // 5. Archive under the original file name.
      const destination = archivePathFor(input.root, note.path);

      try {
        await tools.fs.writeFile({ path: destination, content: transformed });
      } catch (error) {
        // 6. Record the failure and carry on with the next note.
        recordFailure(failures, note.path, error);
        if (failures.length > input.maxFailures) {
          abandoned = true;
          break;
        }
        continue;
      }

      // 7. Record the migrated note in the knowledge graph.
      await tools.memory.createEntities({
        entities: [
          { name: note.path, entityType: "note", observations: [destination] },
        ],
      });
      migrated += 1;
    }
  }

  // 8. Write the report and return the outcome.
  const report = buildReport(migrated, skipped, failures, abandoned);
  await tools.fs.writeFile({ path: `${input.root}/migration-report.md`, content: report });

  return { migrated, skipped, failed: failures.length, abandoned };
}

function parseMarkdownPaths(searchResult: string): string[] {
  return searchResult
    .split("\n")
    .map((line) => line.trim())

function chunk(items: string[], size: number): string[][] {
  const effectiveSize = Math.max(1, Math.floor(size));
  const chunks: string[][] = [];
  for (let index = 0; index < items.length; index += effectiveSize) {
    chunks.push(items.slice(index, index + effectiveSize));
  }
  return chunks;
}

// readMultipleFiles returns one combined string where every file appears as
// `<path>:<text>` and sections are separated by `\n---\n`. Failed reads come
// back as `<path>:Error: ...`. Rebuild one entry per requested path so a file
// the reader could not open still surfaces as that note's own failure.
function parseBatchNotes(requestedPaths: string[], raw: string): ParsedNote[] {
  const textByPath = new Map<string, string>();

  for (const section of raw.split("\n---\n")) {
    const trimmedSection = section.trim();
    if (trimmedSection.length === 0) {
      continue;
    }
    const splitAt = trimmedSection.indexOf(":");
    if (splitAt < 1) {
      continue;
    }
    const path = trimmedSection.slice(0, splitAt).trim();
    const text = trimmedSection.slice(splitAt + 1).trim();
    if (path.length > 0) {
      textByPath.set(path, text);
    }
  }

  return requestedPaths.map((path) => {
    const text = textByPath.get(path);
    if (text === undefined) {
      return { path, text: "", error: "the batch read did not return this file" };
    }
    if (text.startsWith("Error:")) {
      return { path, text: "", error: text };
    }
    return { path, text, error: null };
  });
}
