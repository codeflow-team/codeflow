function parsePaths(content: string): string[] {
  return content
    .split("\n")
    .map((line) => line.trim())
    .filter((line) => line.length > 0)
    .map((line) => line.replace(/^\[(FILE|DIR)\]\s*/, ""));
}

/** Best-effort byte size from the metadata text, falling back to source length. */
function fileSizeBytes(infoContent: string, textContent: string): number {
  const match = infoContent.match(/size["']?\s*[:=]\s*"?(\d+)/i);
  return match ? Number(match[1]) : textContent.length;
}

/** large when over 200 bytes, else documented when it carries a JSDoc block, else plain. */
function gradeModule(sizeBytes: number, text: string): Grade {
  if (sizeBytes > 200) {
    return "large";
  }
  if (text.includes("/**")) {
    return "documented";
  }
  return "plain";
}

function recordFailure(failures: string[], key: string, error: unknown) {
  failures.push(`${key}: ${String(error)}`);
}

  }
  return counts;
}
