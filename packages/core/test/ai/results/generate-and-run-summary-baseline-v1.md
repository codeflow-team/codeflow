  region: string;

function parseListing(content: string): string[] {
  return content
    .split("\n")
    .map((line) => line.trim())
    .filter((line) => line.startsWith("[FILE]"))
    .map((line) => line.slice("[FILE]".length).trim());
}

function isDropFile(name: string): boolean {
  return name.startsWith("drop-") && name.endsWith(".csv");
}

function dropFilesIn(names: string[]): string[] {
  return names.filter(isDropFile);
}

function dataLinesOf(text: string): string[] {
  return text
    .split("\n")
    .map((line) => line.trim())
    .slice(1)
    .filter((line) => line.length > 0);
}

function splitRow(line: string): { id: string; region: string; rawAmount: string } {
  const fields = line.split(",");
  return {
    id: fields[0] ?? "",
    region: fields[1] ?? "",
    rawAmount: fields[2] ?? "",
  };
}

function rejectReasonFor(region: string, rawAmount: string): string | null {
  if (region.trim() === "") {
    return "empty-region";
  }
  if (rawAmount.trim() === "" || Number.isNaN(Number(rawAmount))) {
    return "amount-not-a-number";
  }
  return null;
}

function recordReject(rejects: Reject[], id: string, reason: string): void {
  rejects.push({ id, reason });
}

function toRow(id: string, region: string, rawAmount: string): Row {
  return { id, region, amount: Number(rawAmount) };
}

function emptyTotals(): Map<string, number> {
  return new Map<string, number>();
}

function addToTotals(totals: Map<string, number>, region: string, amount: number): void {
  const current = totals.get(region) ?? 0;
  totals.set(region, current + amount);
}

function renderLedger(totals: Map<string, number>): string {
  const lines = ["# Ledger", ""];
  for (const [region, total] of totals) {
    lines.push(`${region}: ${total.toFixed(2)}`);
  }
  return lines.join("\n");
}

function renderRejects(rejects: Reject[]): string {
  const lines = ["# Rejects", ""];
  for (const reject of rejects) {
    lines.push(`${reject.id}: ${reject.reason}`);
  }
  return lines.join("\n");
}

  input: { root: string; maxRows: number },
  tools: Tools
) {

  const fileNames = parseListing(listing.content);
  const drops = dropFilesIn(fileNames);

  if (drops.length === 0) {
    return { folded: 0, reason: "no-drops" };
  }

  // drop-north.csv is expected to be missing; the read below counts it as a missing region.
  const filesToFold = ["drop-north.csv", ...drops];
  const totals = emptyTotals();
  const rejects: Reject[] = [];
  let folded = 0;
  let rejected = 0;
  let missing = 0;

  for (const fileName of filesToFold) {
    if (folded >= input.maxRows) {
      break;
    }

    let text: string;
    try {
      const file = await tools.fs.readTextFile({
        path: `${input.root}/${fileName}`,
      });
      text = file.content;
    } catch {
      missing += 1;
      continue;

    for (const line of rows) {
      if (folded >= input.maxRows) {
        break;

      const parts = splitRow(line);
      const reason = rejectReasonFor(parts.region, parts.rawAmount);

      if (reason !== null) {
        recordReject(rejects, parts.id, reason);
        rejected += 1;
        continue;
      }

      const row = toRow(parts.id, parts.region, parts.rawAmount);
      addToTotals(totals, row.region, row.amount);
      folded += 1;
    }
  }

  const ledgerBody = renderLedger(totals);
  const rejectsBody = renderRejects(rejects);
  await Promise.all([
    tools.fs.writeFile({ path: `${input.root}/ledger.md`, content: ledgerBody }),
    tools.fs.writeFile({ path: `${input.root}/rejects.md`, content: rejectsBody }),
  ]);

  return { folded, rejected, missing };
}

type DocStatus = "fresh" | "stale" | "missing";

interface ModuleRecord {
  module: string;
  status: DocStatus;
}

function parseModulePaths(result: string): string[] {
  return result
    .split("\n")
    .map((line) => line.trim())
    .filter((line) => line.endsWith(".ts"));
}

function documentationModules(paths: string[]): string[] {
  return paths.filter((path) => !path.endsWith(".test.ts"));
}

function docPathFor(root: string, modulePath: string): string {
  const fileName = modulePath.split("/").pop() ?? modulePath;
  const baseName = fileName.replace(/\.ts$/, "");
  return `${root}/docs/${baseName}.md`;
}

// Tolerates epoch-millisecond timestamps and ISO date strings in the metadata.
function modifiedTimeOf(fileInfo: string): number {
  const line = fileInfo
    .split("\n")
    .find((candidate) => /modif/i.test(candidate));
  if (!line) {
    return 0;
  }
  const epoch = line.match(/\d{12,}/);
  if (epoch) {
    return Number(epoch[0]);
  }
  const parsed = Date.parse(line.slice(line.indexOf(":") + 1).trim());
  return Number.isNaN(parsed) ? 0 : parsed;
}

function isDocStale(moduleInfo: string, docInfo: string): boolean {
  return modifiedTimeOf(docInfo) < modifiedTimeOf(moduleInfo);
}

function recordStatus(records: ModuleRecord[], module: string, status: DocStatus) {
  records.push({ module, status });
}

function countStatus(records: ModuleRecord[], status: DocStatus): number {
  return records.filter((record) => record.status === status).length;
}

function renderReport(records: ModuleRecord[]): string {
  const lines = records.map((record) => `- ${record.module}: ${record.status}`);
  return ["# Documentation freshness", "", ...lines, ""].join("\n");
}

export default async function flow(
  input: { root: string; maxModules: number },
  tools: Tools
) {
  const searchResult = await tools.fs.searchFiles({
    path: input.root,
    pattern: "*.ts",
  });

  const matches = parseModulePaths(searchResult.content);
  const modules = documentationModules(matches);

  if (modules.length === 0) {
    return { audited: 0 };
  }

  const records: ModuleRecord[] = [];
  let audited = 0;

  for (const module of modules) {
    if (audited >= input.maxModules) {
      break;
    }
    audited += 1;

    const doc = docPathFor(input.root, module);
    let status: DocStatus = "missing";

    try {
      const [moduleInfo, docInfo] = await Promise.all([
        tools.fs.getFileInfo({ path: module }),
        tools.fs.getFileInfo({ path: doc }),
      ]);

      if (isDocStale(moduleInfo.content, docInfo.content)) {
        status = "stale";
      } else {
        status = "fresh";
      }
    } catch {
      status = "missing";
    }

    recordStatus(records, module, status);

    if (status === "stale") {
      await tools.memory.createEntities({
        entities: [
          { name: doc, entityType: "stale-doc", observations: [module] },
        ],
      });
    }
  }

  const report = renderReport(records);
  await tools.fs.writeFile({
    path: `${input.root}/freshness.md`,
    content: report,
  });

  const staleCount = countStatus(records, "stale");
  const missingCount = countStatus(records, "missing");

  return { audited, stale: staleCount, missing: missingCount };
}

function parseSearchResults(content: string): string[] {
  const paths: string[] = [];
  for (const line of content.split("\n")) {
    const trimmed = line.trim();
    if (trimmed.startsWith("FILE: ")) {
      const path = trimmed.slice("FILE: ".length).trim();
      if (path !== "") {
        paths.push(path);
      }
    } else if (trimmed.endsWith(".ts") && !trimmed.includes(" ")) {
      paths.push(trimmed);
    }
  }
  return paths;
}

function countLines(source: string): number {
  return source.split("\n").length;
}

function extractExports(source: string): string[] {
  const names: string[] = [];
  const seen = new Set<string>();
  const declarations =
    /export\s+(?:default\s+)?(?:async\s+)?(?:function|class|interface|type|enum|const|let|var)\s+([A-Za-z0-9_$]+)/g;
  for (const match of source.matchAll(declarations)) {
    const name = match[1];
    if (!seen.has(name)) {
      seen.add(name);
      names.push(name);
    }
  }
  const exportLists = /export\s*\{([^}]*)\}/g;
  for (const match of source.matchAll(exportLists)) {
    for (const part of match[1].split(",")) {
      const halves = part.split(/\s+as\s+/);
      const name = halves[halves.length - 1].trim();
      if (name !== "" && !seen.has(name)) {
        seen.add(name);
        names.push(name);
      }
    }
  }
  return names;
}

function foldRows(
  content: string,
  budget: number,
  totals: Map<string, number>,
  rejects: RejectEntry[]
): FoldResult {
  const rows = content
    .split("\n")
    .map((row) => row.trim())
    .filter((row) => row.length > 0);

  let folded = 0;
  let rejected = 0;

  for (const row of rows.slice(1)) {
    if (folded >= budget) {
      break;
    }

    const columns = row.split(",");
    const id = (columns[0] ?? "").trim();
    const region = (columns[1] ?? "").trim();
    const amountRaw = (columns[2] ?? "").trim();

      rejected += 1;
      rejects.push({ id, reason: "empty-region" });
      continue;

      rejected += 1;
      rejects.push({ id, reason: "amount-not-a-number" });
      continue;
    }

    const amount = Number(amountRaw);
    totals.set(region, (totals.get(region) ?? 0) + amount);
    folded += 1;
  }

  return { folded, rejected };
}
