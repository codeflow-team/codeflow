interface ModuleRecord {
  path: string;
  grade: Grade;
}

function parsePaths(content: string): string[] {
  return content
    .split("\n")
    .map((line) => line.trim())
    .filter((line) => line.length > 0);
}

function countNonEmptyLines(text: string): number {
  return text.split("\n").filter((line) => line.trim().length > 0).length;
}

function capModules(paths: string[], max: number): string[] {
  return paths.slice(0, max);
}

function parseSizeBytes(info: string): number {
  const labelled = info.match(/size["']?\s*[:=]\s*(\d+)/i);
  if (labelled !== null) {
    return Number(labelled[1]);
  }
  const bytes = info.match(/(\d+)\s*bytes?\b/i);
  if (bytes !== null) {
    return Number(bytes[1]);
  }
  const first = info.match(/\d+/);
  return first === null ? 0 : Number(first[0]);
}

function gradeModule(sizeBytes: number, text: string): Grade {
  if (sizeBytes > 200) {
    return "large";
  }
  if (text.includes("/**")) {
    return "documented";
  }
  return "plain";
}

function recordModule(records: ModuleRecord[], path: string, grade: Grade): void {
  records.push({ path, grade });
}

function recordFailure(failures: string[], path: string, error: unknown): void {
  failures.push(`${path}: ${String(error)}`);
}

function renderAudit(
  records: ModuleRecord[],
  failures: string[],
  thoughtCount: number
): string {
  const moduleLines = records.map((record) => `- ${record.path}: ${record.grade}`);
  const failureLines = failures.map((failure) => `- ${failure}`);
  return [
    "# Repository Audit",
    "",
    "## modules",
    "",

    "",
    "## unreadable",
    "",

    "",

    ""
  ].join("\n");
}

    large: gradeCounts.large,
    documented: gradeCounts.documented,
    plain: gradeCounts.plain,
    thoughtsRecorded: thoughts.length
  };

/** Turns the raw searchFiles output into absolute module paths. */
function parseModulePaths(content: string): string[] {
  const trimmed = content.trim();
  if (trimmed.startsWith("[")) {
    try {
      const parsed: unknown = JSON.parse(trimmed);
      if (Array.isArray(parsed)) {
        return parsed.filter(
          (entry): entry is string => typeof entry === "string" && entry.endsWith(".ts")
        );
      }
    } catch {
      // Not JSON — fall through to line parsing below.
    }
  }
  return trimmed
    .split("\n")
    .map((line) => line.trim())
    .filter((line) => line.length > 0)
    .filter((line) => !/^\d+\s+matches?:/i.test(line))
    .map((line) => line.replace(/^\[(FILE|DIR)\]\s*/i, ""))
    .filter((line) => line.endsWith(".ts"));
}
